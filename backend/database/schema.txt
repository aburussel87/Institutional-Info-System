CREATE FUNCTION assign_random_provosts_to_halls() RETURNS void
    LANGUAGE plpgsql
    AS $$
DECLARE
    h RECORD;
    t_id INT;
BEGIN
    FOR h IN SELECT hall_id FROM Hall LOOP
        SELECT teacher_id INTO t_id
        FROM Teacher
        WHERE teacher_id NOT IN (
            SELECT teacher_id FROM Provost
        )
        ORDER BY RANDOM()
        LIMIT 1;
        INSERT INTO Provost (teacher_id, hall_id, assigned_on, resigned_on)
        VALUES (t_id, h.hall_id, CURRENT_DATE, NULL);
    END LOOP;
END;
$$;

CREATE FUNCTION get_basic_hall_info_for_provost(p_teacher_id integer) RETURNS TABLE(hall_id integer, name character varying, capacity integer, location character varying, total_assignments bigint, resident_count bigint, non_resident_count bigint)
    LANGUAGE plpgsql
    AS $$
BEGIN
    RETURN QUERY
    SELECT
        h.hall_id,
        h.name,
        h.capacity,
        h.location,
        COALESCE(sa.total, 0) AS total_assignments,
        COALESCE(sa.resident_count, 0) AS resident_count,
        COALESCE(sa.non_resident_count, 0) AS non_resident_count
    FROM hall h
    LEFT JOIN (
        SELECT
            ha.hall_id,
            COUNT(*) AS total,
            COUNT(*) FILTER (WHERE ha.resident = TRUE) AS resident_count,
            COUNT(*) FILTER (WHERE ha.resident = FALSE) AS non_resident_count
        FROM hallassignment ha
        GROUP BY ha.hall_id
    ) sa
    ON sa.hall_id = h.hall_id
    WHERE h.hall_id = (
        SELECT p.hall_id
        FROM provost p
        WHERE p.teacher_id = p_teacher_id
          AND p.resigned_on IS NULL
        LIMIT 1
    );
END;
$$;

-- Name: get_class_routine(semester, character varying, integer); Type: FUNCTION; Schema: public; Owner: system
--

CREATE FUNCTION get_class_routine(p_semester semester, p_academic_session character varying, p_department_id integer) RETURNS TABLE(academic_session character varying, day_of_week character varying, start_time time without time zone, end_time time without time zone, section_type character varying, course_id character varying, course_title character varying, room_id character varying, room_type roomtype, teacher_name character varying)
    LANGUAGE plpgsql
    AS $$
BEGIN
    RETURN QUERY
    SELECT
        cs.academic_session,
        cs.day_of_week,
        cs.start_time,
        cs.end_time,
        cs.section_type,
        cs.course_id,
        c.title AS course_title,
        cs.room_id,
        r.room_type,
        u.username AS teacher_name
    FROM ClassSchedule cs
    JOIN Course c
        ON cs.course_id = c.course_id
    JOIN Room r
        ON cs.room_id = r.room_id
    LEFT JOIN SubjectAllocation sa
        ON sa.course_id = cs.course_id
       AND sa.academic_session = cs.academic_session
       AND (sa.section_type = cs.section_type OR sa.section_type = 'All')
    LEFT JOIN Teacher t
        ON sa.teacher_id = t.teacher_id
    LEFT JOIN "User" u
        ON t.teacher_id = u.user_id
    WHERE cs.academic_session = (SELECT r.academic_session 
          FROM registration_log r
          ORDER BY r.log_id DESC 
          LIMIT 1)
      AND c.semester = p_semester
      AND c.department_id = p_department_id
      AND (cs.section_type IN ('A', 'A1', 'A2', 'All'))
    ORDER BY cs.day_of_week, cs.start_time;
END;
$$;

-- Name: get_department_data(integer); Type: FUNCTION; Schema: public; Owner: system
--

CREATE FUNCTION get_department_data(p_department_id integer) RETURNS jsonb
    LANGUAGE plpgsql
    AS $$
DECLARE
    result JSONB;
BEGIN
    result := jsonb_build_object(
        'teachers', (
            SELECT COALESCE(jsonb_agg(
                jsonb_build_object(
                    'teacher_id', t.teacher_id,
                    'department_id', t.department_id,
                    'username', u.username,
                    'email', u.email,
                    'gender', u.gender,
                    'mobile', u.phone,
                    'photo', u.photo,
                    'designation', t.designation
                )
            ), '[]'::jsonb)
            FROM teacher t
            JOIN "User" u ON t.teacher_id = u.user_id
            WHERE t.department_id = p_department_id AND u.is_active = TRUE
        ),
        
        'students', (
            SELECT COALESCE(jsonb_agg(
                jsonb_build_object(
                    'student_id', s.student_id,
                    'username', u.username,
                    'email', u.email,
                    'gender', u.gender,
                    'mobile', u.phone,
                    'photo', u.photo,
                    'semester', s.current_semester,
                    'cgpa', g.cgpa,
                    'credits', g.total_credit
                )
            ), '[]'::jsonb)
            FROM student s
            JOIN "User" u ON s.student_id = u.user_id
            JOIN gradesheet g ON g.student_id = s.student_id
            WHERE s.department_id = p_department_id AND u.is_active = TRUE
        ),

        'courses', (
            SELECT COALESCE(jsonb_agg(
                jsonb_build_object(
                    'course_id', c.course_id,
                    'title', c.title,
                    'offered_for', c.department_id,
                    'offered_by', c.offered_by,
                    'credits', c.credit_hours,
                    'semester', c.semester
                )
            ), '[]'::jsonb)
            FROM course c
            WHERE (c.offered_by = p_department_id OR c.department_id = p_department_id)
              AND c.still_offered = TRUE
        ),

        'subject_allocations', (
            SELECT COALESCE(jsonb_agg(
                jsonb_build_object(
                    'teacher_id', sa.teacher_id,
                    'teacher_name', u.username,
                    'course_id', sa.course_id,
                    'section_type', sa.section_type,
                    'semester', c.semester
                )
            ), '[]'::jsonb)
            FROM subjectallocation sa
            JOIN teacher t ON t.teacher_id = sa.teacher_id
            JOIN "User" u ON u.user_id = t.teacher_id
            JOIN course c ON c.course_id = sa.course_id
            WHERE t.department_id = p_department_id
              AND sa.academic_session = (
                  SELECT academic_session 
                  FROM registration_log 
                  ORDER BY log_id DESC 
                  LIMIT 1
              )
            AND c.offered_by = p_department_id
        )
    );

    RETURN result;
END;
$$;


ALTER FUNCTION get_department_data(p_department_id integer) OWNER TO system;

-- Name: get_notifications(integer, userrole); Type: FUNCTION; Schema: public; Owner: system
--

CREATE FUNCTION get_notifications(p_uid integer, p_role userrole) RETURNS TABLE(notification_id integer, title character varying, message text, created_by character varying, created_at timestamp without time zone, student_id integer, teacher_id integer, department_id integer, course_id character varying, hall_id integer, semester_id semester, pdf bytea)
    LANGUAGE plpgsql
    AS $$
BEGIN
  IF p_role = 'Student' THEN
    RETURN QUERY
    SELECT
      n.notification_id,
      n.title,
      n.message,
      u.username,
      n.created_at,
      n.student_id,
      n.teacher_id,
      n.department_id,
      n.course_id,
      n.hall_id,
      n.semester_id,
      n.pdf
    FROM notification n
    JOIN student s ON s.student_id = p_uid
    LEFT JOIN "User" u ON u.user_id::VARCHAR = n.created_by
    WHERE
      (n.student_id IS NULL OR n.student_id = s.student_id)
      AND (n.department_id IS NULL OR n.department_id = s.department_id)
      AND (n.hall_id IS NULL OR n.hall_id = s.hall_id)
      AND (n.semester_id IS NULL OR n.semester_id = s.current_semester)
      AND (
        n.course_id IS NULL
        OR (
          n.teacher_id IS NULL AND
          n.course_id IN (
            SELECT e.course_id
            FROM enrollment e
            WHERE e.student_id = s.student_id
              AND e.semester = s.current_semester
          )
        )
        OR (
          n.teacher_id IS NOT NULL AND
          n.course_id IN (
            SELECT e.course_id
            FROM enrollment e
            LEFT JOIN subjectallocation sa ON sa.course_id = e.course_id AND sa.teacher_id = n.teacher_id
            WHERE e.student_id = s.student_id
              AND e.semester = s.current_semester
              AND sa.section_type IN (e.section_type, 'All')
          )
        )
      )
      AND NOT (n.teacher_id IS NOT NULL AND n.course_id IS NULL)
    ORDER BY n.created_at DESC;

  ELSIF p_role = 'Teacher' THEN
    RETURN QUERY
    SELECT
      n.notification_id,
      n.title,
      n.message,
      u.username,
      n.created_at,
      n.student_id,
      n.teacher_id,
      n.department_id,
      n.course_id,
      n.hall_id,
      n.semester_id,
      n.pdf
    FROM notification n
    LEFT JOIN "User" u ON u.user_id::VARCHAR = n.created_by
    WHERE
      n.teacher_id = p_uid
      OR n.department_id = (
        SELECT t.department_id FROM teacher t WHERE t.teacher_id = p_uid
      )
      OR n.course_id IN (
        SELECT sa.course_id
        FROM subjectallocation sa
        WHERE sa.teacher_id = p_uid
      )
    ORDER BY n.created_at DESC;

  ELSIF p_role = 'Admin' THEN
    RETURN QUERY
    SELECT
      n.notification_id,
      n.title,
      n.message,
      u.username,
      n.created_at,
      n.student_id,
      n.teacher_id,
      n.department_id,
      n.course_id,
      n.hall_id,
      n.semester_id,
      n.pdf
    FROM notification n
    LEFT JOIN "User" u ON u.user_id::VARCHAR = n.created_by
    ORDER BY n.created_at DESC;

  ELSE
    RAISE EXCEPTION 'Invalid role: %', p_role;
  END IF;
END;
$$;

CREATE FUNCTION get_teacher_context(p_teacher_id integer, p_session character varying) RETURNS json
    LANGUAGE plpgsql
    AS $$

    result JSON;

        'course', (
            SELECT json_agg(sa)
            FROM subjectallocation sa
            WHERE sa.teacher_id = p_teacher_id
              AND sa.academic_session = (SELECT academic_session 
          FROM registration_log 
          ORDER BY log_id DESC 
          LIMIT 1)
        ),

        -- hall students if teacher is provost

                'student_name', u.username,

            JOIN "User" u ON u.user_id = s.student_id
            WHERE p.teacher_id = p_teacher_id
        ),

                'student_id', s.student_id,
                'student_name', u.username,
                'department_id', s.department_id,
                'current_semester', s.current_semester,
                'advisor_id', s.advisor_id
            ))
            FROM student s
            JOIN "User" u ON u.user_id = s.student_id
            WHERE s.advisor_id = p_teacher_id
        ),

        -- teachers of department if teacher is HOD
        'teacher', (
            SELECT json_agg(json_build_object(
                'teacher_id', t.teacher_id,
                'teacher_name', u.username,
                'department_id', t.department_id
            ))
            FROM head_of_department hod

            JOIN "User" u ON u.user_id = t.teacher_id
            WHERE hod.teacher_id = p_teacher_id

        'deptcourse', (
            SELECT json_agg(c)
            FROM head_of_department hod
            JOIN course c ON c.department_id = hod.department_id
            WHERE hod.teacher_id = p_teacher_id and c.still_offered is not NULL
        ),

        -- department students if teacher is HOD
        'deptstudent', (
            SELECT json_agg(json_build_object(
                'student_id', s.student_id,
                'student_name', u.username,
                'department_id', s.department_id,

            ))
            FROM head_of_department hod
            JOIN student s ON s.department_id = hod.department_id
            JOIN "User" u ON u.user_id = s.student_id
            WHERE hod.teacher_id = p_teacher_id
        )

    ) INTO result;

    RETURN result;

END;

--
-- Name: get_user_notifications(integer, userrole); Type: FUNCTION; Schema: public; Owner: system
--

CREATE FUNCTION get_user_notifications(p_uid integer, p_role userrole) RETURNS TABLE(notification_id integer, title character varying, message text, created_by character varying, created_at timestamp without time zone, student_id integer, teacher_id integer, department_id integer, course_id character varying, hall_id integer, semester_id semester, pdf bytea)
    LANGUAGE plpgsql
    AS $$
BEGIN
  IF p_role = 'Student' THEN
    RETURN QUERY
    SELECT n.notification_id, n.title, n.message, n.created_by, n.created_at,
           n.student_id, n.teacher_id, n.department_id, n.course_id, n.hall_id, n.semester_id, n.pdf
    FROM notification n
    WHERE n.student_id = p_uid
       OR n.department_id = (
            SELECT s.department_id FROM student s WHERE s.student_id = p_uid
         )
       OR n.hall_id = (
            SELECT s.hall_id FROM student s WHERE s.student_id = p_uid
         )
       OR n.semester_id = (
            SELECT s.current_semester FROM student s WHERE s.student_id = p_uid
         )
       OR n.course_id IN (
            SELECT e.course_id
            FROM enrollment e
            JOIN student s ON e.student_id = s.student_id
            WHERE e.student_id = p_uid AND e.semester = s.current_semester
         )
    ORDER BY n.created_at DESC;

  ELSIF p_role = 'Teacher' THEN
    RETURN QUERY
    SELECT n.notification_id, n.title, n.message, n.created_by, n.created_at,
           n.student_id, n.teacher_id, n.department_id, n.course_id, n.hall_id, n.semester_id,n.pdf
    FROM notification n
    WHERE n.teacher_id = p_uid
       OR n.department_id = (
            SELECT t.department_id FROM teacher t WHERE t.teacher_id = p_uid
         )
       OR n.course_id IN (
            SELECT sa.course_id
            FROM subjectallocation sa
            WHERE sa.teacher_id = p_uid
         )
    ORDER BY n.created_at DESC;

  ELSIF p_role = 'Admin' THEN
    RETURN QUERY
    SELECT n.notification_id, n.title, n.message, n.created_by, n.created_at,
           n.student_id, n.teacher_id, n.department_id, n.course_id, n.hall_id, n.semester_id,n.pdf
    FROM notification n
    ORDER BY n.created_at DESC;

  ELSE
    RAISE EXCEPTION 'Invalid role: %', p_role;
  END IF;
END;
$$;

CREATE FUNCTION pay_student_fee(p_student_fee_id integer, p_date timestamp without time zone DEFAULT NULL::timestamp without time zone) RETURNS TABLE(already_paid boolean, msg text)
    LANGUAGE plpgsql
    AS $$
DECLARE
    is_paid BOOL;
BEGIN
  SELECT (status = 'Paid') INTO is_paid
  FROM StudentFee
  WHERE student_fee_id = p_student_fee_id;

  IF NOT FOUND THEN
    RETURN QUERY SELECT FALSE, format('No StudentFee record found with ID %s', p_student_fee_id);
    RETURN;
  END IF;

  IF is_paid THEN
    RETURN QUERY SELECT TRUE, format('StudentFee ID %s is already marked as Paid', p_student_fee_id);
    RETURN;
  END IF;

  UPDATE StudentFee
  SET
    status = 'Paid',
    paid_on = p_date
  WHERE student_fee_id = p_student_fee_id;

  RETURN QUERY SELECT FALSE, format('StudentFee ID %s updated to Paid', p_student_fee_id);
END;
$$;

--
-- Name: update_gradesheet(); Type: FUNCTION; Schema: public; Owner: system
--

CREATE FUNCTION update_gradesheet() RETURNS boolean
    LANGUAGE plpgsql
    AS $$
DECLARE
    total_credits_val FLOAT := 0;
    cgpa_val FLOAT := 0;
    student_total INT;
    weighted_sum FLOAT;
    student_rec RECORD;
    semester_rec RECORD;
    course_rec RECORD;
BEGIN 
    FOR student_rec IN 
        SELECT s.student_id, s.current_semester 
        FROM student s 
        JOIN "User" u ON u.user_id = s.student_id 
        WHERE u.is_active = TRUE
        ORDER BY s.department_id ASC, s.student_id ASC
    LOOP
        -- Reset totals per student
        total_credits_val := 0;
        weighted_sum := 0;

        FOR semester_rec IN 
            SELECT DISTINCT semester 
            FROM courseresult 
            WHERE student_id = student_rec.student_id
        LOOP
            IF semester_rec.semester < student_rec.current_semester THEN
                FOR course_rec IN 
                    SELECT cr.grade_point, c.credit_hours 
                    FROM courseresult cr 
                    JOIN course c ON c.course_id = cr.course_id 
                    WHERE cr.student_id = student_rec.student_id 
                      AND cr.semester = semester_rec.semester
                LOOP
                    total_credits_val := total_credits_val + course_rec.credit_hours;
                    weighted_sum := weighted_sum + (course_rec.credit_hours * course_rec.grade_point);
                END LOOP;
            END IF;
        END LOOP;

        IF total_credits_val > 0 THEN
            cgpa_val := weighted_sum / total_credits_val;
        ELSE
            cgpa_val := 0;
        END IF;
        UPDATE gradesheet
        SET cgpa = cgpa_val, total_credit = total_credits_val
        WHERE student_id = student_rec.student_id;
        
    END LOOP;

    RETURN TRUE;
END;
$$;


ALTER FUNCTION update_gradesheet() OWNER TO system;
